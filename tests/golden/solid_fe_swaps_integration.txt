units metal

pair_style eam/fs

run 0

group g1 type 1
group g2 type 2
variable count1 equal count(g1)
variable count2 equal count(g2)
run 0
fix f1 all nve

fix ff2 g2 ti/spring 10.0 100 100 function 2

fix f5 all print 10 "${step} ${lambda} ${swapatt} ${swapacc}" title "# step lambda swaps_attempted swaps_accepted" screen no file swap.fe.forward_1.dat

run 1000
